cmake_minimum_required(VERSION 3.21)
project(padswipe LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_BUILD_TYPE Release)
find_package(Qt6 6.8 REQUIRED COMPONENTS Core)
set(E ${ENGINE})
file(GLOB SRC ${E}/src/*.cpp)
add_library(engine STATIC ${SRC})
target_include_directories(engine PUBLIC ${E}/include ${E}/src)
target_link_libraries(engine PUBLIC Qt6::Core)
add_executable(padswipe-decode main.cpp)
target_link_libraries(padswipe-decode PRIVATE engine)
